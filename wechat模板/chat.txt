        };

        // 2. 获取地址栏 uid 参数
        const urlParams = new URLSearchParams(location.search);
        const uid = urlParams.get("uid");
        const data = friendData[uid];

        // 3. 渲染页面标题、头像、聊天记录
        document.getElementById("chatName").innerText = data.name;
        const msgBox = document.getElementById("msgBox");
        let html = "";
        data.records.forEach(item => {
            if(item.type === "other"){
                // 对方头像增加点击跳转个人资料页 profile.html
                html += `
                    <div class="msg-row other">
                        <div class="msg-avatar" onclick="location.href='profile.html?uid=${uid}'">
                            <img src="${data.avatar}">
                        </div>
                        <div class="msg-text">${item.text}</div>
                    </div>
                `;
            }else{
                // 自己的头像无点击事件
                html += `
                    <div class="msg-row self">
                        <div class="msg-avatar">
                            <img src="https://picsum.photos/id/237/40/40">
                        </div>
                        <div class="msg-text">${item.text}</div>
                    </div>
                `;
            }
        });
        msgBox.innerHTML = html;
        // 打开页面自动滚动到最新消息底部
        msgBox.scrollTop = msgBox.scrollHeight;
